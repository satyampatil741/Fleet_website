const SHEET_NAME = 'Fleet'; function doGet(e) { const callback = e.parameter.callback; const action = e.parameter.action || 'getAll'; let result; try { if (action === 'getAll') result = { success: true, data: getAllRows() }; else if (action === 'addRow') result = addRow(JSON.parse(e.parameter.data)); else if (action === 'updateRow') result = updateRow(JSON.parse(e.parameter.data)); else if (action === 'deleteRow') result = deleteRow(e.parameter.id); else result = { success: false, error: 'Unknown action' }; } catch(err) { result = { success: false, error: err.toString() }; } const out = callback ? ContentService.createTextOutput(callback+'('+JSON.stringify(result)+')') .setMimeType(ContentService.MimeType.JAVASCRIPT) : ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); return out; } function doPost(e) { try { const body = JSON.parse(e.postData.contents); const { action, data, id } = body; let result; if (action === 'addRow') result = addRow(data); else if (action === 'updateRow') result = updateRow(data); else if (action === 'deleteRow') result = deleteRow(id); else result = { success: false, error: 'Unknown action' }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } const KEYS = ['id','date','regno','cluster','district','make','model','dm','adm','odo','bddate','bdtime','complaint','action','towing','towvendor','workshop','status','approval','estimate','advance','supp','negotiated','invoice','offdays','remarks','logs']; function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); return ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME); } function getAllRows() { const sheet = getSheet(); const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return []; return rows.slice(1).map((row, i) => { const obj = {}; KEYS.forEach((k, j) => { obj[k] = String(row[j] || ''); }); if (!obj.id) obj.id = 'ROW_' + (i + 2); return obj; }); } function addRow(data) { const sheet = getSheet(); const id = 'ROW_' + Date.now(); data.id = id; sheet.appendRow(KEYS.map(k => data[k] || '')); return { success: true, id }; } function updateRow(data) { const sheet = getSheet(); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(data.id)) { KEYS.forEach((k, j) => sheet.getRange(i+1, j+1).setValue(data[k] || '')); return { success: true }; } } return { success: false, error: 'Row not found' }; } function deleteRow(id) { const sheet = getSheet(); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(id)) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false, error: 'Row not found' }; }
